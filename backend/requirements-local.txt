# Local-install-only dependencies. Railway's nixpacks build never reads
# this file (only requirements.txt), so anything here has zero hosted
# footprint by construction — no is_hosted check needed at install time.
#
# playwright: powers the Element Picker feature (services/element_picker.py).
# Used purely as a CDP *client* via connectOverCDP to attach to a Chrome the
# user already has running — never downloads or launches its own Chromium
# when used this way, so `playwright install` is not required for this
# feature (only `pip install playwright`).
#
# Floor is 1.60, NOT 1.44: connect_over_cdp()'s `no_defaults` option — which
# element_picker.py now always requests — was added in Playwright 1.60
# (https://playwright.dev/python/docs/api/class-browsertype#browser-type-connect-over-cdp-option-no-defaults).
# Without it, real Chrome (the user's actual daily-driver browser, not a
# Playwright-launched one) rejects the CDP `Browser.setDownloadBehavior`
# call Playwright issues by default, and connect() fails with:
#   "Protocol error (Browser.setDownloadBehavior): Browser context
#    management is not supported."
# (Confirmed via official Playwright docs + microsoft/playwright issues
# #15370, #30383, #41661 — same root cause reported against every non-
# Playwright-launched CDP target.) element_picker.py also falls back
# gracefully with a `_dlog` if an older Playwright is already installed,
# but this pin is the actual fix — reinstall via
# `pip install -r backend/requirements-local.txt` to pick it up.
playwright>=1.60

# Official MCP client (https://github.com/modelcontextprotocol/python-sdk).
# Local installs only. Do not add this to requirements.txt: mcp 2.x requires
# pydantic>=2.12 and sse-starlette>=3, and requirements.txt pins
# pydantic==2.10.3 and sse-starlette==2.1.3. A local install therefore runs
# a newer pydantic than Railway. The app imports mcp lazily and still boots
# when this package is missing.
# Not the mcp[cli] extra.
mcp>=2,<3
